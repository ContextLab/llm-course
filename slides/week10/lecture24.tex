\begin{frame}[shrink=10]{Testing Reproducibility 🔄}
\begin{alertblock}{Critical Step: Test in Clean Environment}
Before submitting, open a \textbf{brand new} Colab session and run your notebook from top to bottom. This is what graders will do!
\end{alertblock}

\vspace{0.3cm}

\textbf{Common reproducibility issues:}
\begin{enumerate}\setlength{\itemsep}{0pt}
    \item \textbf{Missing installations}: Did you forget a \texttt{!pip install}?
    \item \textbf{File paths}: Are you referencing files that only exist on your machine?
    \item \textbf{API keys}: Did you hard-code keys? (Use environment variables or instructions)
    \item \textbf{Cell order}: Does notebook work when run sequentially?
    \item \textbf{Out-of-order execution}: Did you run cells out of order during development?
    \item \textbf{Runtime requirements}: Does it need a GPU? Does it exceed Colab limits?
\end{enumerate}

\vspace{0.3cm}

\begin{protip}
Use "Runtime → Restart and run all" frequently during development!
\end{protip}
\end{frame}

\section{Common Pitfalls}

\begin{frame}[shrink=10]{Common Pitfalls to Avoid ⚠️}
\textbf{Technical pitfalls:}
\begin{itemize}\setlength{\itemsep}{0pt}
    \item ❌ \textbf{Scope creep}: Trying to do too much at the last minute
    \item ❌ \textbf{No baselines}: Results without comparison are hard to interpret
    \item ❌ \textbf{Cherry-picking}: Only showing best results, hiding failures
    \item ❌ \textbf{Ignoring errors}: Sweeping bugs under the rug instead of fixing
    \item ❌ \textbf{Last-minute testing}: Discovering critical bugs right before deadline
\end{itemize}

\vspace{0.3cm}

\textbf{Presentation pitfalls:}
\begin{itemize}\setlength{\itemsep}{0pt}
    \item ❌ \textbf{Too much detail}: Getting lost in implementation minutiae
    \item ❌ \textbf{No motivation}: Jumping straight to methods without context
    \item ❌ \textbf{Unclear visuals}: Tiny fonts, cluttered slides, confusing plots
    \item ❌ \textbf{Going over time}: Practice and time yourself!
    \item ❌ \textbf{Unbalanced team}: One person doing all the talking
\end{itemize}
\end{frame}

\begin{frame}{When Things Don't Work 🔧}
\small
\begin{block}{Research Reality}
Not everything will work perfectly. That's \textbf{completely normal} in research!
\end{block}

\vspace{0.3cm}

\textbf{If experiments fail or results are negative:}
\begin{itemize}\setlength{\itemsep}{0pt}
    \item \textbf{Don't panic!} Negative results are still valuable
    \item \textbf{Analyze why}: What went wrong? What did you learn?
    \item \textbf{Be honest}: Discuss failures and limitations openly
    \item \textbf{Show process}: Document what you tried and why
    \item \textbf{Suggest improvements}: What would you do differently?
\end{itemize}

\vspace{0.3cm}

\begin{protip}
Projects are graded on process, rigor, and critical thinking—not just positive results. A thoughtful analysis of why something didn't work can be just as valuable as showing success!
\end{protip}
\end{frame}

\begin{frame}[shrink=10]{Getting Unstuck 🚀}
\textbf{If you're stuck on a technical issue:}
\begin{enumerate}\setlength{\itemsep}{0pt}
    \item \textbf{Read the error message carefully}: What exactly is breaking?
    \item \textbf{Google the error}: Someone has likely encountered it before
    \item \textbf{Check documentation}: Are you using the API correctly?
    \item \textbf{Simplify}: Create minimal example that reproduces the issue
    \item \textbf{Use GenAI tools}: Claude, ChatGPT can help debug
    \item \textbf{Ask for help}: Discord, peers, instructor
\end{enumerate}

\vspace{0.3cm}

\textbf{If you're stuck on project direction:}
\begin{enumerate}\setlength{\itemsep}{0pt}
    \item \textbf{Review your proposal}: What was your original plan?
    \item \textbf{Prioritize}: What's essential vs. nice-to-have?
    \item \textbf{Scope down}: Better to do less well than more poorly
    \item \textbf{Talk it through}: Explain the issue to a teammate or peer
    \item \textbf{Consult instructor}: Get feedback on revised approach
\end{enumerate}
